Overview

agdb is a high-performance, persistent database engine written in Zig, designed for hybrid search (full-text + vector), multi-tenancy, and low-latency operations.

https://agdb.replit.app

All API endpoints are prefixed with /v1. Authenticated endpoints require a Bearer token in the Authorization header.

Base URL: https://agdb.replit.app
All requests and responses use application/json.

Authentication

All authenticated endpoints require a Bearer token in the Authorization header.

Authorization: Bearer agdb_xxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxx
Keep your API key secret. It grants full access to your tenant database.

Python Client

No dependencies beyond the standard library. Copy the class below into your project.

# No pip install needed — pure stdlib
"""agdb Cloud Python client — copy & paste, no dependencies"""
import urllib.request, urllib.error, json
from typing import Any, Optional

class AgdbError(Exception):
    def __init__(self, status: int, message: str):
        super().__init__(message)
        self.status = status

class AgdbClient:
    def __init__(self, base_url: str, api_key: Optional[str] = None):
        self.base_url = base_url.rstrip("/")
        self.api_key  = api_key

    def _req(self, method: str, path: str, body: Any = None, auth: bool = True) -> Any:
        url  = self.base_url + path
        data = json.dumps(body).encode() if body is not None else None
        headers = {"Content-Type": "application/json"}
        if auth and self.api_key:
            headers["Authorization"] = f"Bearer {self.api_key}"
        req = urllib.request.Request(url, data=data, headers=headers, method=method)
        try:
            with urllib.request.urlopen(req, timeout=30) as r:
                return json.loads(r.read().decode())
        except urllib.error.HTTPError as e:
            msg = json.loads(e.read().decode()).get("error", str(e))
            raise AgdbError(e.code, msg)

    # ── Auth ──────────────────────────────────────────────────────────────────

    def register(self, email: str) -> dict:
        """Register a new account. Returns {"tenant_id", "api_key"}."""
        return self._req("POST", "/v1/auth/register", {"email": email}, auth=False)

    def login(self, api_key: str) -> dict:
        """Verify an API key and return tenant info."""
        self.api_key = api_key
        return self._req("POST", "/v1/auth/login", {"api_key": api_key}, auth=False)

    # ── Account ───────────────────────────────────────────────────────────────

    def get_tenant(self) -> dict:
        """Get current tenant info."""
        return self._req("GET", "/v1/tenant")

    def delete_account(self) -> dict:
        """Permanently delete account and all data."""
        return self._req("DELETE", "/v1/account")

    # ── API Keys ──────────────────────────────────────────────────────────────

    def list_keys(self) -> list:
        return self._req("GET", "/v1/apikeys")["keys"]

    def rotate_key(self, key_id: str) -> dict:
        """Rotate (replace) an API key. The old key is immediately invalidated."""
        return self._req("POST", f"/v1/apikeys/{key_id}/rotate")

    def revoke_key(self, key_id: str) -> dict:
        return self._req("DELETE", f"/v1/apikeys/{key_id}")

    # ── Records ───────────────────────────────────────────────────────────────

    def insert(self, record: dict, db: str = "documents") -> dict:
        """Insert a record. Returns {"id": ...}."""
        return self._req("POST", f"/v1/databases/{db}/records", {"record": record})

    def get(self, record_id: int, db: str = "documents") -> dict:
        return self._req("GET", f"/v1/databases/{db}/records/{record_id}")

    def delete(self, record_id: int, db: str = "documents") -> dict:
        return self._req("DELETE", f"/v1/databases/{db}/records/{record_id}")

    # ── Search ────────────────────────────────────────────────────────────────

    def search(self, query: str, top_k: int = 10, db: str = "documents") -> list:
        """Hybrid full-text + vector search."""
        resp = self._req("POST", f"/v1/databases/{db}/search",
                         {"query": query, "top_k": top_k})
        return resp.get("results", resp)

    # ── Sandbox ───────────────────────────────────────────────────────────────

    def sandbox_status(self) -> dict:
        return self._req("GET", "/v1/sandbox")

    def sandbox_start(self) -> dict:
        return self._req("POST", "/v1/sandbox/start")

    def sandbox_stop(self) -> dict:
        return self._req("POST", "/v1/sandbox/stop")

    def sandbox_restart(self) -> dict:
        return self._req("POST", "/v1/sandbox/restart")

    # ── Stats ─────────────────────────────────────────────────────────────────

    def stats(self, db: str = "documents") -> dict:
        return self._req("GET", f"/v1/databases/{db}/stats")

    # ── Health ────────────────────────────────────────────────────────────────

    def health(self) -> dict:
        return self._req("GET", "/v1/health", auth=False)


# ── Usage example ──────────────────────────────────────────────────────────

Zig Client

Pure Zig, no external dependencies. Uses std.http.Client (Zig 0.14+).

// agdb Cloud Zig client — Zig 0.14+
// Add to your build.zig: no external deps needed
const std = @import("std");

pub const AgdbError = error {
    Unauthorized, NotFound, Conflict, ServerError, BadResponse,
};

pub const AgdbClient = struct {
    allocator: std.mem.Allocator,
    base_url:  []const u8,
    api_key:   ?[]const u8,
    client:    std.http.Client,

    pub fn init(allocator: std.mem.Allocator, base_url: []const u8, api_key: ?[]const u8) AgdbClient {
        return .{
            .allocator = allocator,
            .base_url  = base_url,
            .api_key   = api_key,
            .client    = .{ .allocator = allocator },
        };
    }

    pub fn deinit(self: *AgdbClient) void { self.client.deinit(); }

    // Low-level request helper
    fn request(
        self: *AgdbClient, method: std.http.Method,
        path: []const u8, body: ?[]const u8, auth: bool,
    ) ![]u8 {
        const url = try std.fmt.allocPrint(self.allocator, "{s}{s}", .{ self.base_url, path });
        defer self.allocator.free(url);
        const uri = try std.Uri.parse(url);

        var extra = std.ArrayList(std.http.Header).init(self.allocator);
        defer extra.deinit();
        try extra.append(.{ .name = "Content-Type", .value = "application/json" });
        if (auth) if (self.api_key) |k| {
            const bearer = try std.fmt.allocPrint(self.allocator, "Bearer {s}", .{k});
            defer self.allocator.free(bearer);
            try extra.append(.{ .name = "Authorization", .value = bearer });
        };

        var hdr_buf: [8192]u8 = undefined;
        var req = try self.client.open(method, uri, .{
            .server_header_buffer = &hdr_buf,
            .extra_headers = extra.items,
        });
        defer req.deinit();

        if (body) |b| { req.transfer_encoding = .{ .content_length = b.len }; }
        try req.send();
        if (body) |b| try req.writeAll(b);
        try req.finish();
        try req.wait();

        return switch (req.response.status) {
            .ok, .created, .no_content => req.reader().readAllAlloc(self.allocator, 16 * 1024 * 1024),
            .unauthorized => AgdbError.Unauthorized,
            .not_found    => AgdbError.NotFound,
            .conflict     => AgdbError.Conflict,
            else          => AgdbError.ServerError,
        };
    }

    // ── Auth ──────────────────────────────────────────────────────────────────

    pub fn register(self: *AgdbClient, email: []const u8) ![]u8 {
        const body = try std.fmt.allocPrint(self.allocator, "{{\\"email\\":\\"{s}\\"}}", .{email});
        defer self.allocator.free(body);
        return self.request(.POST, "/v1/auth/register", body, false);
    }

    pub fn login(self: *AgdbClient, api_key: []const u8) ![]u8 {
        self.api_key = api_key;
        const body = try std.fmt.allocPrint(self.allocator, "{{\\"api_key\\":\\"{s}\\"}}", .{api_key});
        defer self.allocator.free(body);
        return self.request(.POST, "/v1/auth/login", body, false);
    }

    // ── Records ───────────────────────────────────────────────────────────────

    pub fn insert(self: *AgdbClient, record_json: []const u8, db: []const u8) ![]u8 {
        const path = try std.fmt.allocPrint(self.allocator, "/v1/databases/{s}/records", .{db});
        defer self.allocator.free(path);
        const body = try std.fmt.allocPrint(self.allocator, "{{\\"record\\":{s}}}", .{record_json});
        defer self.allocator.free(body);
        return self.request(.POST, path, body, true);
    }

    pub fn get(self: *AgdbClient, id: u64, db: []const u8) ![]u8 {
        const path = try std.fmt.allocPrint(self.allocator, "/v1/databases/{s}/records/{d}", .{ db, id });
        defer self.allocator.free(path);
        return self.request(.GET, path, null, true);
    }

    pub fn delete(self: *AgdbClient, id: u64, db: []const u8) ![]u8 {
        const path = try std.fmt.allocPrint(self.allocator, "/v1/databases/{s}/records/{d}", .{ db, id });
        defer self.allocator.free(path);
        return self.request(.DELETE, path, null, true);
    }

    // ── Search ────────────────────────────────────────────────────────────────

    pub fn search(self: *AgdbClient, query: []const u8, top_k: u32, db: []const u8) ![]u8 {
        const path = try std.fmt.allocPrint(self.allocator, "/v1/databases/{s}/search", .{db});
        defer self.allocator.free(path);
        const body = try std.fmt.allocPrint(self.allocator,
            "{{\\"query\\":\\"{s}\\",\\"top_k\\":{d}}}", .{ query, top_k });
        defer self.allocator.free(body);
        return self.request(.POST, path, body, true);
    }

    // ── Health ────────────────────────────────────────────────────────────────

    pub fn health(self: *AgdbClient) ![]u8 {
        return self.request(.GET, "/v1/health", null, false);
    }
};

// ── Usage example ──────────────────────────────────────────────────────────
pub fn main() !void {
    var gpa = std.heap.GeneralPurposeAllocator(.{}){};
    defer _ = gpa.deinit();
    const alloc = gpa.allocator();

    var client = AgdbClient.init(alloc, "https://agdb.replit.app", null);
    defer client.deinit();

    // Register
    const account_json = try client.register("alice@example.com");
    defer alloc.free(account_json);
    std.debug.print("Account: {s}\n", .{account_json});

    // Insert
    client.api_key = "agdb_yourkey";
    const r = try client.insert(
        "{\"title\":\"Zig is fast\",\"body\":\"Tight machine code.\"}", "documents");
    defer alloc.free(r);

    // Search
    const results = try client.search("fast compiled", 5, "documents");
    defer alloc.free(results);
    std.debug.print("Search: {s}\n", .{results});
}
